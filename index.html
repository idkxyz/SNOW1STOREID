export default { async fetch(request, env, ctx) { const url = new URL(request.url); const fileId = url.searchParams.get('id') || '33u48ez7xbbp4iy'; const mediafireUrl = `https://www.mediafire.com/file/${fileId}/file`; try { const response = await fetch(mediafireUrl, { headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36' } }); const html = await response.text(); // Regex lebih spesifik: mencari tag dengan class "popsok" // dan mengambil nilai href-nya. Tanda kutip bisa tunggal atau ganda. const match = html.match(/]+class="[^"]*popsok[^"]*"[^>]+href="([^"]+)"/); // Jika tidak ketemu dengan class "popsok", coba pola alternatif const fallbackMatch = html.match(/href="(https:\/\/download[^"]+\.mediafire\.com[^"]+)"/); const directLink = match ? match[1] : (fallbackMatch ? fallbackMatch[1] : null); if (directLink) { return Response.redirect(directLink, 302); } else { // Jika tetap gagal, kembalikan ke halaman MediaFire biasa return Response.redirect(mediafireUrl, 302); } } catch (error) { return Response.redirect(mediafireUrl, 302); } } };
