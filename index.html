export default { async fetch(request, env, ctx) { const url = new URL(request.url); // Mengambil ID file MediaFire const fileId = url.searchParams.get('id') || '33u48ez7xbbp4iy'; const mediafireUrl = `https://www.mediafire.com/file/${fileId}/file`; try { // ========================================== // METODE 1: MENGGUNAKAN API (Paling Stabil) // ========================================== const apiRes = await fetch(`https://mediafire-url.vercel.app/api/media`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: fileId }) }); if (apiRes.ok) { const data = await apiRes.json(); if (data && data.link) { // Jika API berhasil, langsung download return Response.redirect(data.link, 302); } } // ========================================== // METODE 2: SCRAPING HTML (Cadangan) // ========================================== const response = await fetch(mediafireUrl, { headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36' } }); if (response.ok) { const html = await response.text(); // Cari link yang berawalan download...mediafire.com const match = html.match(/href="(https:\/\/download[^"]+mediafire\.com[^"]+)"/i); if (match && match[1]) { return Response.redirect(match[1], 302); } } // ========================================== // METODE 3: FALLBACK (Jika semua gagal) // ========================================== return Response.redirect(mediafireUrl, 302); } catch (error) { // Jika terjadi error sistem, kembalikan ke MediaFire return Response.redirect(mediafireUrl, 302); } } };
