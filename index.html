export default { async fetch(request, env, ctx) { const url = new URL(request.url); const fileId = url.searchParams.get('id') || '33u48ez7xbbp4iy'; const mediafireUrl = `https://www.mediafire.com/file/${fileId}/file`; try { const response = await fetch(mediafireUrl, { headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36' } }); const html = await response.text(); // REGEX BARU: Mencari link yang mengandung "download" dan "mediafire.com" // Ini jauh lebih stabil dan tidak terpengaruh perubahan tag HTML const match = html.match(/href="(https:\/\/download[^"]+mediafire\.com[^"]+)"/i); if (match && match[1]) { // Jika link ditemukan, langsung redirect ke file APK return Response.redirect(match[1], 302); } else { // Jika gagal, arahkan ke halaman MediaFire biasa return Response.redirect(mediafireUrl, 302); } } catch (error) { return Response.redirect(mediafireUrl, 302); } } };
